# Frontend (Macacha) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir `frontend/`, la interfaz de chat en Next.js que consume el agente conversacional ya implementado (`POST /chat` SSE, `GET /sesiones/{id}/mensajes`), con sesión anónima, streaming visual, fuentes citadas y manejo de errores; y habilitar CORS en el backend para que el navegador pueda llamarlo directo.

**Architecture:** App Next.js (App Router, TypeScript, Tailwind) sin librerías de estado adicionales — un hook (`useChatStream`) maneja el `fetch` + lectura manual del `ReadableStream` de `POST /chat`, parseando líneas `data: {...}` en eventos tipados (`texto`/`fin`/`error`) que van actualizando el estado de la conversación en memoria. La sesión es un UUID generado en el cliente y persistido en `localStorage`.

**Tech Stack:** Next.js 15 (App Router), React 19, TypeScript 5, Tailwind CSS 3, Vitest (único test automatizado: el parseo de eventos SSE). Backend: FastAPI `CORSMiddleware` (ya viene con FastAPI, sin dependencia nueva).

## Global Constraints

- Sin librerías de estado adicionales (React Query, SWR, Zustand, Redux) — `useState`/hooks propios.
- `session_id` generado con `crypto.randomUUID()`, persistido en `localStorage` bajo la clave `macacha_session_id`.
- URL del backend vía `NEXT_PUBLIC_API_URL` (default `http://localhost:8000` si no está seteada).
- Contrato de eventos SSE de `POST /chat`: `{"tipo":"texto","delta":str}`, `{"tipo":"fin","fuentes":[{"tramite_id","nombre_oficial","fuente_url"}]}`, `{"tipo":"error","mensaje":str}` — ya implementados en el backend, no se modifican.
- `GET /sesiones/{id}/mensajes` devuelve `[{"rol","contenido","creado_en"}]`, `[]` si la sesión no existe.
- CORS del backend permite como origen `http://localhost:3000` por defecto, configurable vía `FRONTEND_ORIGIN`.
- Única prueba automatizada del frontend: el parseo de eventos SSE (`parsearLineasSSE`) con Vitest. El resto de la UI se verifica manualmente en el navegador.
- Frontend y backend se levantan con comandos separados (`npm run dev` / `uvicorn` + `docker compose`), no se agrega el frontend a `docker-compose.yml`.
- Todo el código del frontend vive bajo `frontend/` en el repo `macacha`.

```typescript
"use client";

import { useEffect, useState } from "react";

const CLAVE_SESSION_ID = "macacha_session_id";

export function useSession(): { sessionId: string | null } {
  const [sessionId, setSessionId] = useState<string | null>(null);

  useEffect(() => {
    const existente = localStorage.getItem(CLAVE_SESSION_ID);
    if (existente) {
      setSessionId(existente);
      return;
    }
    const nuevo = crypto.randomUUID();
    localStorage.setItem(CLAVE_SESSION_ID, nuevo);
    setSessionId(nuevo);
  }, []);

  return { sessionId };
}
```

```typescript
"use client";

import { useEffect, useState } from "react";
import { obtenerHistorial } from "../lib/api";

export type Fuente = {
  tramite_id: string;
  nombre_oficial: string;
  fuente_url: string | null;
};

export type Mensaje = {
  rol: "user" | "assistant";
  contenido: string;
  fuentes?: Fuente[];
  error?: boolean;
};

export type EventoSSE =
  | { tipo: "texto"; delta: string }
  | { tipo: "fin"; fuentes: Fuente[] }
  | { tipo: "error"; mensaje: string };

export function parsearLineasSSE(texto: string): EventoSSE[] {
  return texto
    .split("\n\n")
    .filter((linea) => linea.startsWith("data: "))
    .map((linea) => JSON.parse(linea.slice("data: ".length)) as EventoSSE);
}

const BASE_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

export function useChatStream(sessionId: string) {
  const [mensajes, setMensajes] = useState<Mensaje[]>([]);
  const [enviando, setEnviando] = useState(false);

  useEffect(() => {
    obtenerHistorial(sessionId).then((historial) => {
      setMensajes(
        historial.map((m) => ({ rol: m.rol, contenido: m.contenido }))
      );
    });
  }, [sessionId]);

  function aplicarEvento(evento: EventoSSE) {
    setMensajes((prev) => {
      const copia = [...prev];
      const ultimo = copia[copia.length - 1];
      if (evento.tipo === "texto") {
        copia[copia.length - 1] = {
          ...ultimo,
          contenido: ultimo.contenido + evento.delta,
        };
      } else if (evento.tipo === "fin") {
        copia[copia.length - 1] = { ...ultimo, fuentes: evento.fuentes };
      } else if (evento.tipo === "error") {
        copia[copia.length - 1] = {
          ...ultimo,
          contenido: evento.mensaje,
          error: true,
        };
      }
      return copia;
    });
  }

  async function enviarMensaje(texto: string) {
    setMensajes((prev) => [
      ...prev,
      { rol: "user", contenido: texto },
      { rol: "assistant", contenido: "" },
    ]);
    setEnviando(true);

    const respuesta = await fetch(`${BASE_URL}/chat`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ session_id: sessionId, mensaje: texto }),
    });

    const lector = respuesta.body?.getReader();
    const decodificador = new TextDecoder();
    let acumulado = "";

    if (lector) {
      while (true) {
        const { done, value } = await lector.read();
        if (done) break;
        acumulado += decodificador.decode(value, { stream: true });

        const partes = acumulado.split("\n\n");
        acumulado = partes.pop() ?? "";

        if (partes.length > 0) {
          for (const evento of parsearLineasSSE(partes.join("\n\n"))) {
            aplicarEvento(evento);
          }
        }
      }
    }

    setEnviando(false);
  }

  return { mensajes, enviando, enviarMensaje };
}
```

```tsx
"use client";

import { useState } from "react";

export function ChatInput({
  disabled,
  onEnviar,
}: {
  disabled: boolean;
  onEnviar: (texto: string) => void;
}) {
  const [texto, setTexto] = useState("");

  function enviar() {
    const limpio = texto.trim();
    if (!limpio || disabled) return;
    onEnviar(limpio);
    setTexto("");
  }

  return (
    <div className="flex gap-2 border-t border-gray-200 p-4">
      <textarea
        className="flex-1 resize-none rounded border border-gray-300 p-2"
        rows={2}
        value={texto}
        disabled={disabled}
        onChange={(e) => setTexto(e.target.value)}
        onKeyDown={(e) => {
          if (e.key === "Enter" && !e.shiftKey) {
            e.preventDefault();
            enviar();
          }
        }}
        placeholder="Escribí tu consulta sobre un trámite..."
      />
      <button
        className="rounded bg-blue-600 px-4 py-2 text-white disabled:opacity-50"
        disabled={disabled || !texto.trim()}
        onClick={enviar}
      >
        Enviar
      </button>
    </div>
  );
}
```

```tsx
  const { mensajes, enviando, enviarMensaje } = useChatStream(sessionId);

  return (
    <main className="mx-auto flex h-screen max-w-2xl flex-col">
      <header className="border-b border-gray-200 p-4">
        <h1 className="text-lg font-semibold">Macacha</h1>
        <p className="text-sm text-gray-500">
          Asistente de trámites — Provincia de Salta
        </p>
      </header>
      <div className="flex-1 space-y-3 overflow-y-auto p-4">
        {mensajes.map((mensaje, indice) => (
          <ChatMessage
            key={indice}
            mensaje={mensaje}
            onReintentar={
              mensaje.error
                ? () => {
                    const anterior = mensajes[mensajes.length - 2];
                    if (anterior) enviarMensaje(anterior.contenido);
                  }
                : undefined
            }
          />
        ))}
        {enviando && <p className="text-sm text-gray-400">escribiendo…</p>}
      </div>
      <ChatInput disabled={enviando} onEnviar={enviarMensaje} />
    </main>
  );
}
```
